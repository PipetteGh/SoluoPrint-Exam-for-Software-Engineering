create table public.companies (
  id uuid not null default extensions.uuid_generate_v4 (),
  name character varying(255) not null,
  email character varying(255) null,
  phone character varying(50) null,
  address text null,
  logo_url text null,
  currency character varying(10) null default 'GHS'::character varying,
  currency_symbol character varying(5) null default '¢'::character varying,
  timezone character varying(100) null default 'Africa/Accra'::character varying,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint companies_pkey primary key (id)
) TABLESPACE pg_default;

create trigger update_companies_updated_at BEFORE
update on companies for EACH row
execute FUNCTION update_updated_at ();

create table public.job_list (
  id uuid not null default gen_random_uuid (),
  company_id uuid not null,
  customer_id uuid not null,
  services jsonb null default '[]'::jsonb,
  description text null,
  notes text null,
  status text null default 'Pending'::text,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  images jsonb null default '[]'::jsonb,
  constraint job_list_pkey primary key (id),
  constraint job_list_company_id_fkey foreign KEY (company_id) references companies (id) on delete CASCADE,
  constraint job_list_customer_id_fkey foreign KEY (customer_id) references customers (id) on delete CASCADE,
  constraint job_list_status_check check (
    (
      status = any (
        array[
          'Pending'::text,
          'In Progress'::text,
          'Ready'::text,
          'Converted'::text,
          'Cancelled'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_job_list_company_id on public.job_list using btree (company_id) TABLESPACE pg_default;

create index IF not exists idx_job_list_customer_id on public.job_list using btree (customer_id) TABLESPACE pg_default;

create index IF not exists idx_job_list_status on public.job_list using btree (status) TABLESPACE pg_default;
